<script>
  const openBtn = document.getElementById('openTrailer');
  const popup = document.getElementById('trailerPopup');
  const closeBtn = document.getElementById('closePopup');
  const trailerVideo = document.getElementById('trailerVideo');

  const trailerURL = "https://www.youtube.com/embed/"; // trailer

  openBtn.onclick = () => {
    popup.style.display = "flex";          // munculkan popup
    trailerVideo.src = trailerURL + "?autoplay=1";  // mulai video otomatis
  };

  closeBtn.onclick = () => {
    popup.style.display = "none";
    trailerVideo.src = ""; // stop video
  };

  // Tutup popup kalau klik di luar kotak
  window.onclick = (e) => {
    if (e.target === popup) {
      popup.style.display = "none";
      trailerVideo.src = "";
    }
  };
</script>
